
import 'package:flutter/material.dart';

void main() {
  runApp(const QuizMasterApp());
}

class QuizMasterApp extends StatelessWidget {
  const QuizMasterApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      title: 'Quiz Application',
      theme: ThemeData(
        colorSchemeSeed: Colors.deepPurple,
        useMaterial3: true,
      ),
      home: const HomeScreen(),
    );
  }
}

class Question {
  String question;
  List<String> options;
  int answer;

  Question(this.question, this.options, this.answer);
}

class HomeScreen extends StatefulWidget {
  const HomeScreen({super.key});

  @override
  State<HomeScreen> createState() => _HomeScreenState();
}

class _HomeScreenState extends State<HomeScreen> {
  final Map<String, List<Question>> quizzes = {
    'Flutter': [
      Question('Flutter is developed by?', ['Google', 'Microsoft', 'Apple', 'Meta'], 0),
      Question('Which language is used in Flutter?', ['Java', 'Dart', 'Python', 'C'], 1),
      Question('Which widget is used for layout?', ['Text', 'Container', 'Scaffold', 'All of these'], 3),
    ],
    'Computer Science': [
      Question('What is the brain of a computer?', ['RAM', 'CPU', 'ROM', 'Hard Disk'], 1),
      Question('What does HTML stand for?', ['Hyper Text Markup Language', 'High Text Machine Language', 'Home Tool Markup Language', 'None'], 0),
      Question('Which is an operating system?', ['Google', 'Python', 'Windows', 'HTML'], 2),
    ],
    'General Knowledge': [
      Question('What is the capital of India?', ['Mumbai', 'New Delhi', 'Chennai', 'Hyderabad'], 1),
      Question('How many days are there in a leap year?', ['365', '366', '364', '360'], 1),
      Question('Which planet is known as the Red Planet?', ['Earth', 'Venus', 'Mars', 'Jupiter'], 2),
    ],
  };

  String? selectedCategory;
  int currentQuestion = 0;
  int score = 0;
  int bestScore = 0;
  int? selectedAnswer;
  bool answered = false;

  void startQuiz(String category) {
    setState(() {
      selectedCategory = category;
      currentQuestion = 0;
      score = 0;
      selectedAnswer = null;
      answered = false;
    });
  }

  void checkAnswer(int index) {
    if (answered) return;

    setState(() {
      selectedAnswer = index;
      answered = true;

      if (index ==
          quizzes[selectedCategory]![currentQuestion].answer) {
        score++;
      }
    });
  }

  void nextQuestion() {
    setState(() {
      if (currentQuestion <
          quizzes[selectedCategory]!.length - 1) {
        currentQuestion++;
        selectedAnswer = null;
        answered = false;
      } else {
        if (score > bestScore) bestScore = score;
        selectedCategory = null;
        showResult = true;
      }
    });
  }

  bool showResult = false;

  void addQuestion(String category, String question,
      List<String> options, int answer) {
    setState(() {
      quizzes[category]!.add(
        Question(question, options, answer),
      );
    });
  }

  @override
  Widget build(BuildContext context) {
    if (showResult) {
      return Scaffold(
        appBar: AppBar(title: const Text('Quiz Results')),
        body: Center(
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              const Icon(Icons.emoji_events,
                  size: 90, color: Colors.amber),
              const SizedBox(height: 20),
              const Text('Quiz Completed!',
                  style: TextStyle(
                      fontSize: 28, fontWeight: FontWeight.bold)),
              const SizedBox(height: 15),
              Text('Your Score: $score',
                  style: const TextStyle(fontSize: 24)),
              Text('Best Score: $bestScore',
                  style: const TextStyle(fontSize: 20)),
              const SizedBox(height: 25),
              ElevatedButton(
                onPressed: () {
                  setState(() => showResult = false);
                },
                child: const Text('Back to Home'),
              ),
            ],
          ),
        ),
      );
    }

    if (selectedCategory != null) {
      final questions = quizzes[selectedCategory]!;
      final q = questions[currentQuestion];

      return Scaffold(
        appBar: AppBar(title: Text(selectedCategory!)),
        body: Padding(
          padding: const EdgeInsets.all(20),
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.stretch,
            children: [
              Text(
                'Question ${currentQuestion + 1} of ${questions.length}',
                style: const TextStyle(
                    fontSize: 18, fontWeight: FontWeight.bold),
              ),
              const SizedBox(height: 20),
              LinearProgressIndicator(
                value: (currentQuestion + 1) / questions.length,
              ),
              const SizedBox(height: 30),
              Text(q.question,
                  style: const TextStyle(
                      fontSize: 23, fontWeight: FontWeight.bold)),
              const SizedBox(height: 25),
              ...List.generate(q.options.length, (index) {
                Color? color;

                if (answered && index == q.answer) {
                  color = Colors.green.shade200;
                } else if (answered && index == selectedAnswer) {
                  color = Colors.red.shade200;
                }

                return Container(
                  margin: const EdgeInsets.only(bottom: 12),
                  child: ElevatedButton(
                    style: ElevatedButton.styleFrom(
                      backgroundColor: color,
                      foregroundColor: Colors.black,
                      padding: const EdgeInsets.all(16),
                    ),
                    onPressed: () => checkAnswer(index),
                    child: Text(q.options[index]),
                  ),
                );
              }),
              const Spacer(),
              if (answered)
                ElevatedButton(
                  onPressed: nextQuestion,
                  child: Text(
                    currentQuestion == questions.length - 1
                        ? 'Finish Quiz'
                        : 'Next Question',
                  ),
                ),
            ],
          ),
        ),
      );
    }

    return Scaffold(
      appBar: AppBar(
        title: const Text('Quiz Application'),
        centerTitle: true,
      ),
      body: Padding(
        padding: const EdgeInsets.all(20),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            const SizedBox(height: 15),
            const Icon(Icons.quiz,
                size: 80, color: Colors.deepPurple),
            const Text(
              'Welcome to Quiz Application!',
              textAlign: TextAlign.center,
              style: TextStyle(
                  fontSize: 25, fontWeight: FontWeight.bold),
            ),
            const SizedBox(height: 10),
            const Text(
              'Choose a category and test your knowledge.',
              textAlign: TextAlign.center,
            ),
            const SizedBox(height: 25),
            const Text('Quiz Categories',
                style: TextStyle(
                    fontSize: 20, fontWeight: FontWeight.bold)),
            const SizedBox(height: 15),
            ...quizzes.keys.map((category) {
              return Card(
                child: ListTile(
                  leading: const Icon(Icons.menu_book,
                      color: Colors.deepPurple),
                  title: Text(category),
                  trailing: const Icon(Icons.arrow_forward_ios),
                  onTap: () => startQuiz(category),
                ),
              );
            }),
            const SizedBox(height: 15),
            ElevatedButton.icon(
              onPressed: () {
                Navigator.push(
                  context,
                  MaterialPageRoute(
                    builder: (_) => AddQuestionScreen(
                      categories: quizzes.keys.toList(),
                      onAdd: addQuestion,
                    ),
                  ),
                );
              },
              icon: const Icon(Icons.add),
              label: const Text('Add Your Own Question'),
            ),
            const Spacer(),
            Text('Best Score: $bestScore',
                textAlign: TextAlign.center,
                style: const TextStyle(fontSize: 17)),
          ],
        ),
      ),
    );
  }
}

class AddQuestionScreen extends StatefulWidget {
  final List<String> categories;
  final Function(String, String, List<String>, int) onAdd;

  const AddQuestionScreen({
    super.key,
    required this.categories,
    required this.onAdd,
  });

  @override
  State<AddQuestionScreen> createState() =>
      _AddQuestionScreenState();
}

class _AddQuestionScreenState extends State<AddQuestionScreen> {
  final questionController = TextEditingController();

  final optionControllers =
      List.generate(4, (_) => TextEditingController());

  String? category;
  int correctAnswer = 0;

  @override
  void initState() {
    super.initState();
    category = widget.categories.first;
  }

  @override
  void dispose() {
    questionController.dispose();

    for (final controller in optionControllers) {
      controller.dispose();
    }

    super.dispose();
  }

  void saveQuestion() {
    if (questionController.text.trim().isEmpty ||
        optionControllers.any(
          (controller) => controller.text.trim().isEmpty,
        )) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Please fill all fields'),
        ),
      );
      return;
    }

    widget.onAdd(
      category!,
      questionController.text.trim(),
      optionControllers
          .map((controller) => controller.text.trim())
          .toList(),
      correctAnswer,
    );

    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(
        content: Text('Question added successfully!'),
      ),
    );

    Navigator.pop(context);
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Add Question'),
      ),
      body: ListView(
        padding: const EdgeInsets.all(20),
        children: [
          DropdownButtonFormField<String>(
            initialValue: category,
            decoration: const InputDecoration(
              labelText: 'Select Category',
              border: OutlineInputBorder(),
            ),
            items: widget.categories.map((c) {
              return DropdownMenuItem<String>(
                value: c,
                child: Text(c),
              );
            }).toList(),
            onChanged: (value) {
              setState(() {
                category = value;
              });
            },
          ),

          const SizedBox(height: 20),

          TextField(
            controller: questionController,
            decoration: const InputDecoration(
              labelText: 'Enter Question',
              border: OutlineInputBorder(),
            ),
          ),

          const SizedBox(height: 20),

          const Text(
            'Enter Four Options',
            style: TextStyle(
              fontSize: 18,
              fontWeight: FontWeight.bold,
            ),
          ),

          ...List.generate(4, (index) {
            return Padding(
              padding: const EdgeInsets.symmetric(vertical: 8),
              child: TextField(
                controller: optionControllers[index],
                decoration: InputDecoration(
                  labelText: 'Option ${index + 1}',
                  border: const OutlineInputBorder(),
                ),
              ),
            );
          }),

          const SizedBox(height: 15),

          DropdownButtonFormField<int>(
            initialValue: correctAnswer,
            decoration: const InputDecoration(
              labelText: 'Select Correct Answer',
              border: OutlineInputBorder(),
            ),
            items: List.generate(4, (index) {
              return DropdownMenuItem<int>(
                value: index,
                child: Text('Option ${index + 1}'),
              );
            }),
            onChanged: (value) {
              if (value != null) {
                setState(() {
                  correctAnswer = value;
                });
              }
            },
          ),

          const SizedBox(height: 25),

          ElevatedButton(
            onPressed: saveQuestion,
            child: const Text('Save Question'),
          ),
        ],
      ),
    );
  }
}
